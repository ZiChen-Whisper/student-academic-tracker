AI 生成的 SQL

SELECT s.student_name, AVG(es.score) AS avg_score
FROM student s
JOIN exam_score es ON s.student_id = es.student_id
WHERE es.exam_stage = 'G3'
GROUP BY s.student_id
ORDER BY avg_score DESC
LIMIT 10;

SELECT s.student_name, es.score
FROM student s
JOIN exam_score es ON s.student_id = es.student_id
WHERE es.subject_id = 'SUBJ_MATH' AND es.exam_stage = 'G3'
ORDER BY es.score DESC
LIMIT 10;
